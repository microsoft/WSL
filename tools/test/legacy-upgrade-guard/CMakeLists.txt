cmake_minimum_required(VERSION 3.25)
project(legacy_upgrade_guard_tests LANGUAGES CXX)
if(NOT WIL_INCLUDE_DIR)
    message(FATAL_ERROR "Pass -DWIL_INCLUDE_DIR=<WIL checkout>/include")
endif()
add_executable(guard-tests guard-tests.cpp)
target_compile_features(guard-tests PRIVATE cxx_std_20)
target_include_directories(guard-tests PRIVATE "${WIL_INCLUDE_DIR}" "${CMAKE_CURRENT_LIST_DIR}/../../../src/windows/wslinstaller/exe")
target_link_libraries(guard-tests PRIVATE advapi32)
enable_testing()
add_test(NAME service-upgrade-guard COMMAND guard-tests)
